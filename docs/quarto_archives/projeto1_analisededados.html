jama_colors <- c('#374e55', '#df8f44', '#00a1d5', '#b24745')

labels_desfechos <- list(
  Frutas = "Frutas",
  Legumes_Verduras = "Legumes e Verduras",
  Bebidas_Alcoolicas = "Bebidas Alcoólicas",
  Produtos_Industrializados = "Produtos Industrializados",
  Consumo_agua = "Consumo de Água",
  Embutidos = "Embutidos"
)

grafico_distribuicao <- function(df, col, labels_desfechos = list(), colors = jama_colors) {
    
  col_label <- if (col %in% names(labels_desfechos)) labels_desfechos[[col]] else col
  
  grafico <- ggplot(df, aes(x = .data[[col]], fill = .data[[col]])) +
    geom_bar() +
    stat_count(
      geom = "text",
      aes(label = after_stat(count)),
      vjust = -0.5,
      size = 4,
      color = "black"
    ) +
    scale_fill_manual(values = colors) +
    labs(
      title = paste("Consumo de", col_label),
      x = col_label,
      y = "Contagem"
    ) +
    theme_bw() +
    theme(
      axis.text.x = element_text(angle = 45, hjust = 1),
      legend.position = "none"
    )
  
  return(grafico)
}
